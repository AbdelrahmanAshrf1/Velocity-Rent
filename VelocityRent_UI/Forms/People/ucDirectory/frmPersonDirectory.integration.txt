// 4) ApplyQuery remains simple:
private void ApplyQuery()
{
    IEnumerable<PersonDtotest> query = _persons;
    string search = _txtSearch == null ? string.Empty : _txtSearch.Text.Trim();

    if (!string.IsNullOrWhiteSpace(search))
    {
        query = query.Where(p =>
            Contains(p.FullName, search) ||
            Contains(p.NationalID, search) ||
            Contains(p.Phone, search));
    }

    if (_cmbCity != null && _cmbCity.SelectedIndex > 0 && _cmbCity.SelectedItem != null)
    {
        string city = _cmbCity.SelectedItem.ToString();
        query = query.Where(p => string.Equals(
            p.City, city, StringComparison.OrdinalIgnoreCase));
    }

    bool ascending = _cmbSort == null || _cmbSort.SelectedIndex <= 0;

    query = ascending
        ? query.OrderBy(p => p.FullName)
        : query.OrderByDescending(p => p.FullName);

    // This method now EXISTS on ucDirectory.
    _personGrid.SetData(query.ToList());
}

// 5) Replace PersonRowActionEventArgs with DirectoryRowActionEventArgs.
private void PersonGrid_ActionClicked(
    object sender,
    DirectoryRowActionEventArgs e)
{
    var person = e.GetItem<PersonDtotest>();

    if (e.ActionKey == "View")
        PersonGrid_ViewClicked(person);
    else if (e.ActionKey == "Edit")
        PersonGrid_EditClicked(person);
}

private void PersonGrid_ViewClicked(PersonDtotest person)
{
    MessageBox.Show(
        string.Format("{0}\r\n{1}\r\n{2}\r\n{3}\r\n{4}, {5}",
            person.FullName,
            person.NationalID,
            person.Email,
            person.Phone,
            person.City,
            person.Country),
        "Person Details",
        MessageBoxButtons.OK,
        MessageBoxIcon.Information);
}

private void PersonGrid_EditClicked(PersonDtotest person)
{
    MessageBox.Show(
        "Open your existing person editor here for ID " + person.ID + ".",
        "Edit Person",
        MessageBoxButtons.OK,
        MessageBoxIcon.Information);
}
